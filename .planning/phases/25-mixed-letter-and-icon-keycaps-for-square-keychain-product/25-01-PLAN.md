export function ensureKeycapSequence(raw: unknown): KeycapSlot[] {
  if (raw == null) return [];
  let arr: unknown = raw;
  if (typeof raw === "string") {
    if (raw.trim() === "") return [];
    try { arr = JSON.parse(raw); } catch { return []; }
  }
  if (!Array.isArray(arr)) return [];
  const out: KeycapSlot[] = [];
  for (const s of arr) {
    if (s && typeof s === "object") {
      const o = s as Record<string, unknown>;
      if (o.t === "L" && typeof o.ch === "string" && o.ch.length === 1) out.push({ t: "L", ch: o.ch });
      else if (o.t === "I" && typeof o.id === "string" && o.id) out.push({ t: "I", id: o.id });
    }
  }
  return out;
}
```

/**
 * Human-readable summary for a keycap sequence, e.g.
 *   '"SOUD" + [Alien] + [Skull] (6 keycaps: 4 letters, 2 icons)'
 * Letter runs are quoted; icons are shown by label in brackets. Unknown icon id → id string.
 * Does NOT include the colour tail ("· Magenta base · …") — callers append colour parts as today.
 */
export function buildKeycapSequenceSummary(
  slots: KeycapSlot[],
  iconLabelById: Record<string, string>,
): string {
  const parts: string[] = [];
  let run = "";
  const flush = () => { if (run) { parts.push(`"${run}"`); run = ""; } };
  let letters = 0, icons = 0;
  for (const s of slots) {
    if (s.t === "L") { run += s.ch; letters++; }
    else { flush(); parts.push(`[${iconLabelById[s.id] ?? s.id}]`); icons++; }
  }
  flush();
  const total = slots.length;
  if (total === 0) return "";
  const breakdown = `${total} keycap${total === 1 ? "" : "s"}: ${letters} letter${letters === 1 ? "" : "s"}, ${icons} icon${icons === 1 ? "" : "s"}`;
  return `${parts.join(" + ")} (${breakdown})`;
}
```
