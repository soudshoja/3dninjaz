export function resolveTierWeightKg(unitFieldId: string | null, configValues: Record<string,string>, weightTiers: Record<string,number>): number | null {
  if (!unitFieldId) return null;
  const v = configValues[unitFieldId];
  if (typeof v !== "string" || v.length === 0) return null;
  const grams = weightTiers[String(v.length)];   // ← .length is WRONG for a keycapseq JSON blob
  if (typeof grams !== "number" || !Number.isFinite(grams) || grams < 0) return null;
  return grams / 1000;
}
