```typescript
import { ensureKeycapSequence, type KeycapSlot } from "@/lib/config-fields";

export type KeycapSequenceParts = {
  slots: KeycapSlot[];
  slotCount: number;    // D-12 pricing: letters + icons
  letterCount: number;  // BASE/CLICKER batches (Pitfall 1)
  iconCount: number;    // icon batch
  letters: string[];    // the letter chars in order
  icons: string[];      // the icon ids in order
};

/** Structured-first read. Returns null when the field has no keycap sequence (caller falls back to parseKeychainParts). */
export function parseKeycapSequence(raw: unknown, seqFieldId: string): KeycapSequenceParts | null {
  const cfg = ensureConfigurationData(raw); // reuse the existing helper; import from config-fields
  const slots = ensureKeycapSequence(cfg?.values?.[seqFieldId]);
  if (slots.length === 0) return null;
  const letters = slots.filter((s): s is { t: "L"; ch: string } => s.t === "L").map((s) => s.ch);
  const icons = slots.filter((s): s is { t: "I"; id: string } => s.t === "I").map((s) => s.id);
  return { slots, slotCount: slots.length, letterCount: letters.length, iconCount: icons.length, letters, icons };
}
```
