```typescript
// BEFORE
import { db } from "@/lib/db";
export async function POST(req: NextRequest) {
  const raw = await req.text();
  if (!verifySignature(raw, sig, secret)) return 401;   // HMAC over raw body FIRST — UNCHANGED
  if (!remember(key)) return 200;                        // in-process idempotency
  await db.update(orderShipments)...                     // module singleton
}
// AFTER
import { getTenantContext } from "@/lib/tenant/context";
export async function POST(req: NextRequest) {
  const raw = await req.text();
  if (!verifySignature(raw, sig, secret)) return 401;   // verify BEFORE resolving tenant — UNCHANGED
  const { tenant, db } = await getTenantContext();       // Host-resolved; single mode = singleton
  if (!remember(`${tenant.id}:${key}`)) return 200;      // idempotency key tenant-scoped
  await db.update(orderShipments)...                     // tenant db; DB-level UNIQUE delyvaShipmentId is per-tenant
}
```
